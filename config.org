  "Read a Lisp expression from STR.
Signal an error if the entire string was not used."
  (pcase-let ((`(,expr . ,offset) (read-from-string str)))
    (condition-case ()
        ;; The call to `ignore' suppresses a compiler warning.
        (progn (ignore (read-from-string str offset))
               (error "Can't read whole string"))

           (file (expand-file-name "Query.v" dir))
           members)

          (with-temp-buffer
            (write-region (format "%s\nSearch _ inside %s.\n" requires module) nil file nil 'silent)

(defvar pl-symbol--tex-reverse
  (condition-case err
      (progn
        (require 'cape-char)
        ;; Invert the TeX input method (\\alpha -> α) to get α -> \\alpha,
        ;; reusing the same authoritative data that backs `cape-tex'.
        (let ((fwd (cape-char--translation "TeX" '(?\\)))
              (rev (make-hash-table :test #'equal)))
          (when fwd
            (maphash (lambda (name val)
                       (let ((s (if (stringp val) val (char-to-string val))))
                         (push name (gethash s rev))))
                     fwd))
          rev))
    (error (message "pl-symbol: TeX names unavailable: %S" err)
           (make-hash-table :test #'equal)))
  "Hash of symbol to the LaTeX commands the TeX input method maps onto it.")

(defun pl-symbol--latex-for (sym)
  "LaTeX commands for SYM: manual overrides first, then TeX-derived."
  (delete-dups
   (append (cdr (assoc sym pl-symbol-latex-overrides))
           ;; \\Bbb{N} is redundant with the nicer \\mathbb{N} override.
           (seq-remove (lambda (n) (string-match-p "Bbb" n))
                       (reverse (gethash sym pl-symbol--tex-reverse))))))

(defun pl-symbol-refresh ()
  "Rebuild the lookup tables from `pl-symbol-rules'.
Concept names are registered first and win every collision; letter-only
LaTeX commands are then added where they don't clash.  Warn only about
two concept names that disagree."
  (clrhash pl-symbol-hash)
  (clrhash pl-symbol-doc-hash)
  (clrhash pl-symbol-alias-hash)
  (clrhash pl-symbol-latex-hash)
  ;; Pass 1: concept names (authoritative).
  (pcase-dolist (`(,sym ,desc . ,names) pl-symbol-rules)
    (dolist (name names)
      (let ((key (concat "\\" name)))
        (when-let* ((prev (gethash key pl-symbol-hash))
                    ((not (equal prev sym))))
          (warn "pl-symbol: `%s' maps to both %s and %s" key prev sym))
        (puthash key sym pl-symbol-hash)
        (puthash key desc pl-symbol-doc-hash)))
    (puthash sym (mapcar (lambda (n) (concat "\\" n)) names) pl-symbol-alias-hash))
  ;; Pass 2: LaTeX commands (added only where the key is still free).
  (pcase-dolist (`(,sym ,desc . ,_names) pl-symbol-rules)
    (let ((latex (pl-symbol--latex-for sym)))
      (puthash sym latex pl-symbol-latex-hash)
      (dolist (l latex)
        (when (and (string-match-p "\\`\\\\[A-Za-z]+\\'" l)
                   (not (gethash l pl-symbol-hash)))
          (puthash l sym pl-symbol-hash)
          (puthash l desc pl-symbol-doc-hash))))))

(defconst pl-symbol--zwsp #x200b
  "Zero-width space, appended to keep LaTeX candidate strings unique.")

(defvar pl-symbol--dyn (make-hash-table :test #'equal)
  "Per-call map of candidate string to (INSERT SYM DESC KIND).")

(defun pl-symbol--annotation (cand)
  (pcase (gethash cand pl-symbol--dyn)
    (`(,_ ,sym ,desc :sym) (format "  %s  %s " sym (or desc "")))
    (`(,ins ,_ ,_ :tex)    (format "  %s  LaTeX " ins))))

(defun pl-symbol--docsig (cand)
  (pcase (gethash cand pl-symbol--dyn)
    (`(,_ ,sym ,desc :sym) (format "%s  %s" sym (or desc "")))
    (`(,ins ,_ ,desc :tex) (format "%s  (LaTeX)  %s" ins (or desc "")))))

(defun pl-symbol--doc-buffer (cand)
  "Documentation buffer shown by `corfu-popupinfo-mode'."
  (when-let* ((entry (gethash cand pl-symbol--dyn))
              (sym (nth 1 entry)))
    (with-current-buffer (get-buffer-create " *pl-symbol-doc*")
      (erase-buffer)
      (insert sym "  " (or (nth 2 entry) "") "\n\n"
              "Inserts: " (nth 0 entry)
              (if (eq (nth 3 entry) :tex) "  (LaTeX literal)" "  (symbol)") "\n\n"
              "Names: " (string-join (gethash sym pl-symbol-alias-hash) "  ") "\n"
              "LaTeX: " (let ((l (gethash sym pl-symbol-latex-hash)))
                          (if l (string-join l "  ") "—"))
              "\n\n"
              (mapconcat (lambda (c)
                           (format "U+%04X  %s" c
                                   (or (get-char-code-property c 'name) "")))
                         (string-to-list sym) "\n"))
      (goto-char (point-min))
      (current-buffer))))

(defun pl-symbol--exit (cand status)
  (when-let* (((not (eq status 'exact)))
              (entry (gethash cand pl-symbol--dyn)))
    (delete-region (max (point-min) (- (point) (length cand))) (point))
    (insert (nth 0 entry))))

(defun pl-symbol-capf (&optional interactive)
  "Complete a PL symbol after a \\ prefix, offering the glyph and its LaTeX."
  (interactive (list t))
  (if interactive
      (let (completion-cycle-threshold)
        (cape-interactive #'pl-symbol-capf))
    ;; In TeX buffers \alpha should stay \alpha, not become α.
    (unless (derived-mode-p 'tex-mode 'latex-mode)
      (when (looking-back "\\\\[^ \n\t]*" (pos-bol))
        (let ((beg (match-beginning 0))
              (tok (match-string-no-properties 0))
              (cands nil))
          (clrhash pl-symbol--dyn)
          (maphash
           (lambda (nm sym)
             (when (string-prefix-p tok nm)
               (let ((desc (gethash nm pl-symbol-doc-hash))
                     (i 0))
                 ;; glyph row
                 (puthash nm (list sym sym desc :sym) pl-symbol--dyn)
                 (push nm cands)
                 ;; LaTeX literal rows, kept unique with zero-width spaces
                 (dolist (l (gethash sym pl-symbol-latex-hash))
                   (let ((key (concat nm (make-string (setq i (1+ i))
                                                      pl-symbol--zwsp))))
                     (puthash key (list l sym desc :tex) pl-symbol--dyn)
                     (push key cands))))))
           pl-symbol-hash)
          (when cands
            (list beg (point) (nreverse cands)
                  :annotation-function #'pl-symbol--annotation
                  :company-docsig #'pl-symbol--docsig
                  :company-doc-buffer #'pl-symbol--doc-buffer
                  :exit-function #'pl-symbol--exit
                  :company-kind (lambda (_) 'text)
                  :category 'pl-symbol
                  :exclusive 'no)))))))

    "Return a persp name for PROJECT-ROOT. If another project persp already

  so both projects are disambiguated symmetrically."
    (let* ((root (file-truename (directory-file-name project-root)))
           (leaf (file-name-nondirectory root)))
      (or
       ;; Already have a persp for this exact root — reuse it.
       (cl-some (lambda (name)
                  (let ((p (persp-get-by-name name)))
                    (when (and (perspective-p p)
                               (equal (persp-parameter 'project-root p) root))
                      name)))

       ;; Leaf taken by *another* project persp? Rename it, qualify ours.
       (let* ((existing (persp-get-by-name leaf))
              (existing-root (and (perspective-p existing)
                                  (persp-parameter 'project-root existing))))
         (cond
          ((and existing-root (not (equal existing-root root)))
           (persp-rename (persp-qualified-name existing-root) existing)
           (persp-qualified-name root))

          ((perspective-p existing) (persp-qualified-name root))
